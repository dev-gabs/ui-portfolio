  document.addEventListener('DOMContentLoaded', () => {
    // 1. Dynamic Year
    const yearEl = document.getElementById('current-year');
    if (yearEl) yearEl.textContent = new Date().getFullYear();

    // 2. TWO-STATE NAVIGATION SYSTEM (Upscayl Inspired)
    const navHeader = document.getElementById('nav-header');
    const navContainer = document.getElementById('nav-container');

    const handleScrollNav = () => {
      const isScrolled = window.scrollY > 40;
      if (isScrolled) {
        // STATE B: Detaches smoothly into a floating glass capsule with dotted texture
        navHeader.classList.remove('py-5');
        navHeader.classList.add('py-3');
        navContainer.className = 'max-w-5xl mx-auto flex items-center justify-between transition-all duration-500 ease-out bg-white/75 backdrop-blur-md nav-dotted-texture border border-white/90 shadow-capsule rounded-full px-5 py-2';
      } else {
        // STATE A: Integrated smoothly into the top area canvas. Translucent, airy, no heavy shadow
        navHeader.classList.remove('py-3');
        navHeader.classList.add('py-5');
        navContainer.className = 'max-w-7xl mx-auto flex items-center justify-between transition-all duration-500 ease-out bg-transparent border border-transparent shadow-none px-0 py-0';
      }
    };

    window.addEventListener('scroll', handleScrollNav, { passive: true });
    handleScrollNav(); // initial check

    // 3. WORKSPACE MODES HORIZONTAL TRANSITION & PHYSICAL SLIDER
    const tabTools = document.getElementById('tab-btn-tools');
    const tabProjects = document.getElementById('tab-btn-projects');
    const slidingPill = document.getElementById('sliding-pill');
    const viewTools = document.getElementById('workspace-view-tools');
    const viewProjects = document.getElementById('workspace-view-projects');
    const heroQuickTools = document.getElementById('hero-quick-tools');
    const heroQuickProjects = document.getElementById('hero-quick-projects');
    const mobileDockTools = document.getElementById('mobile-dock-tools');
    const mobileDockProjects = document.getElementById('mobile-dock-projects');
    const inlineSwitch = document.getElementById('switch-to-projects-inline');

    let currentWorkspaceMode = 'tools';

    const setWorkspaceView = (mode) => {
      if (mode === 'tools') {
        currentWorkspaceMode = 'tools';
        // Slider pill positioning
        if (slidingPill) {
          slidingPill.style.left = '4px';
          slidingPill.style.width = '108px';
        }
        tabTools.classList.add('text-white');
        tabTools.classList.remove('text-slate-600');
        tabProjects.classList.remove('text-white');
        tabProjects.classList.add('text-slate-600');

        // Horizontal View Transition
        viewTools.className = 'workspace-view view-active';
        viewProjects.className = 'workspace-view view-hidden-right';

        // Update Hero preview buttons
        if (heroQuickTools && heroQuickProjects) {
          heroQuickTools.className = 'px-3 py-1 rounded-full bg-navy-800 text-white shadow-xs transition';
          heroQuickProjects.className = 'px-3 py-1 rounded-full text-slate-600 hover:text-navy-900 transition';
        }
        // Update Mobile dock buttons
        if (mobileDockTools && mobileDockProjects) {
          mobileDockTools.className = 'px-4 py-2 rounded-full font-bold transition bg-accent-indigo text-white shadow-xs';
          mobileDockProjects.className = 'px-4 py-2 rounded-full font-bold transition text-slate-300 hover:text-white';
        }
      } else {
        currentWorkspaceMode = 'projects';
        // Slider pill positioning
        if (slidingPill) {
          slidingPill.style.left = '112px';
          slidingPill.style.width = '122px';
        }
        tabProjects.classList.add('text-white');
        tabProjects.classList.remove('text-slate-600');
        tabTools.classList.remove('text-white');
        tabTools.classList.add('text-slate-600');

        // Horizontal View Transition
        viewTools.className = 'workspace-view view-hidden-left';
        viewProjects.className = 'workspace-view view-active';

        // Update Hero preview buttons
        if (heroQuickTools && heroQuickProjects) {
          heroQuickProjects.className = 'px-3 py-1 rounded-full bg-navy-800 text-white shadow-xs transition';
          heroQuickTools.className = 'px-3 py-1 rounded-full text-slate-600 hover:text-navy-900 transition';
        }
        // Update Mobile dock buttons
        if (mobileDockTools && mobileDockProjects) {
          mobileDockProjects.className = 'px-4 py-2 rounded-full font-bold transition bg-accent-indigo text-white shadow-xs';
          mobileDockTools.className = 'px-4 py-2 rounded-full font-bold transition text-slate-300 hover:text-white';
        }
      }
    };

    if (tabTools) tabTools.addEventListener('click', () => setWorkspaceView('tools'));
    if (tabProjects) tabProjects.addEventListener('click', () => setWorkspaceView('projects'));
    if (inlineSwitch) inlineSwitch.addEventListener('click', () => setWorkspaceView('projects'));

    if (heroQuickTools) {
      heroQuickTools.addEventListener('click', () => {
        setWorkspaceView('tools');
        document.getElementById('workspace').scrollIntoView({ behavior: 'smooth' });

    }
    if (heroQuickProjects) {
      heroQuickProjects.addEventListener('click', () => {

        document.getElementById('workspace').scrollIntoView({ behavior: 'smooth' });

    if (mobileDockTools) mobileDockTools.addEventListener('click', () => setWorkspaceView('tools'));

    // 4. TWO-WAY TOOL <-> PROJECT INTERACTIVE HIGHLIGHTING
    const highlightTools = (toolIdsString) => {
      if (!toolIdsString) return;
      const ids = toolIdsString.split(',');
      ids.forEach(id => {
        const el = document.getElementById(id.trim());
        if (el) el.classList.add('tool-highlighted');
      });
    };

    const clearHighlightedTools = () => {
      document.querySelectorAll('.tool-highlighted').forEach(el => {
        el.classList.remove('tool-highlighted');
      });
    };

    const featuredProject = document.getElementById('project-featured');
    if (featuredProject) {
      featuredProject.addEventListener('mouseenter', () => {
        highlightTools(featuredProject.getAttribute('data-tools'));

    const secondaryCards = document.querySelectorAll('.project-card-mini');

      card.addEventListener('mouseenter', () => {
        highlightTools(card.getAttribute('data-tools'));
      });
      card.addEventListener('mouseleave', clearHighlightedTools);
    });

    // Inverse interaction: Hovering on active tool badges (Figma, Illustrator, ChatGPT) links back to project
    const activeToolElements = document.querySelectorAll('.symbiosis-active');
    activeToolElements.forEach(tool => {
      tool.addEventListener('mouseenter', () => {
        if (featuredProject) {
          featuredProject.classList.add('ring-2', 'ring-accent-indigo', 'ring-offset-2');
        }
      });
      tool.addEventListener('mouseleave', () => {
        if (featuredProject) {
          featuredProject.classList.remove('ring-2', 'ring-accent-indigo', 'ring-offset-2');
        }
      });
    });

    // 5. PROJECT CATEGORY FILTER BUTTONS
    const filterBtns = document.querySelectorAll('.project-filter-btn');
    filterBtns.forEach(btn => {
      btn.addEventListener('click', () => {
        filterBtns.forEach(b => {
          b.className = 'project-filter-btn px-4 py-1.5 rounded-full text-xs font-semibold bg-white/80 text-slate-600 hover:bg-white border border-slate-200/80 transition';
        });
        btn.className = 'project-filter-btn px-4 py-1.5 rounded-full text-xs font-semibold bg-navy-800 text-white shadow-xs transition';
      });
    });

    // 6. COPY EMAIL INTERACTION
    const copyBtn = document.getElementById('copy-email-btn');
    if (copyBtn) {
      copyBtn.addEventListener('click', () => {
        navigator.clipboard.writeText('hello@gabrielsalvador.design');
        const orig = copyBtn.innerText;
        copyBtn.innerText = 'Copied to clipboard!';
        copyBtn.classList.add('text-accent-indigo');
        setTimeout(() => {
          copyBtn.innerText = orig;
          copyBtn.classList.remove('text-accent-indigo');
        }, 2200);
      });
    }

    // 7. MOUSE-MOVE PARALLAX RESPONSE ON HERO ARTIFACTS
    const heroSection = document.getElementById('hero');
    const artifacts = document.querySelectorAll('.parallax-artifact');
    if (heroSection && artifacts.length > 0) {
      heroSection.addEventListener('mousemove', (e) => {
        const rect = heroSection.getBoundingClientRect();
        const x = (e.clientX - rect.left) / rect.width - 0.5;
        const y = (e.clientY - rect.top) / rect.height - 0.5;

        artifacts.forEach((art, index) => {
          const depth = (index + 1) * 12;
          art.style.transform = `translate(${x * depth}px, ${y * depth}px)`;
        });
      });

      heroSection.addEventListener('mouseleave', () => {
        artifacts.forEach(art => {
          art.style.transform = 'translate(0px, 0px)';
        });
      });
    }
  });
</script>
